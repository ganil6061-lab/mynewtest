  <script>
    class Game2048 {
      constructor() {
        this.size = 4;
        this.grid = [];
        this.score = 0;
        this.bestScore = parseInt(localStorage.getItem('2048_best_score') || '0', 10);
        this.won = false;
        this.keepPlaying = false;
        this.gameOver = false;
        this.history = []; // For Undo functionality

        // UI references
        this.tileLayer = document.getElementById('tile-layer');
        this.scoreEl = document.getElementById('score');
        this.bestScoreEl = document.getElementById('best-score');
        this.scoreFloatContainer = document.getElementById('score-float-container');
        this.undoBtn = document.getElementById('undo-btn');
        this.gameOverModal = document.getElementById('game-over-modal');
        this.winModal = document.getElementById('win-modal');

        this.bestScoreEl.textContent = this.bestScore;
        this.initEventListeners();
        this.restart();
      }

      /* Initialize fresh grid state */
      restart() {
        this.grid = Array.from({ length: this.size }, () => Array(this.size).fill(0));
        this.score = 0;
        this.won = false;
        this.keepPlaying = false;
        this.gameOver = false;
        this.history = [];
        this.updateUndoButton();
        this.gameOverModal.classList.add('hidden');
        this.winModal.classList.add('hidden');

        this.addRandomTile();
        this.addRandomTile();
        this.render();
      }

      /* Save snapshot for Undo */
      saveState() {
        const state = {
          grid: this.grid.map(row => [...row]),
          score: this.score,
          won: this.won,
          keepPlaying: this.keepPlaying
        };
        this.history.push(state);
        if (this.history.length > 10) this.history.shift(); // Keep last 10 moves
        this.updateUndoButton();
      }

      undo() {
        if (this.history.length === 0 || this.gameOver) return;
        const lastState = this.history.pop();
        this.grid = lastState.grid;
        this.score = lastState.score;
        this.won = lastState.won;
        this.keepPlaying = lastState.keepPlaying;
        this.gameOver = false;
        this.gameOverModal.classList.add('hidden');
        this.winModal.classList.add('hidden');
        this.updateUndoButton();
        this.render();
      }

      updateUndoButton() {
        this.undoBtn.disabled = this.history.length === 0;
      }

      addRandomTile() {
        const emptyCells = [];
        for (let r = 0; r < this.size; r++) {
          for (let c = 0; c < this.size; c++) {
            if (this.grid[r][c] === 0) emptyCells.push({ r, c });
          }
        }
        if (emptyCells.length > 0) {
          const { r, c } = emptyCells[Math.floor(Math.random() * emptyCells.length)];
          this.grid[r][c] = Math.random() < 0.9 ? 2 : 4;
          return { r, c, val: this.grid[r][c] };
        }
        return null;
      }

      slide(row) {
        let arr = row.filter(val => val !== 0);
        let points = 0;
        let mergedIndices = [];

        for (let i = 0; i < arr.length - 1; i++) {
          if (arr[i] === arr[i + 1]) {
            arr[i] *= 2;
            points += arr[i];
            arr[i + 1] = 0;
            mergedIndices.push(i);
          }
        }

        arr = arr.filter(val => val !== 0);
        while (arr.length < this.size) {
          arr.push(0);
        }

        return { newRow: arr, points };
      }

      move(direction) {
        if (this.gameOver) return false;

        let moved = false;
        let gainedPoints = 0;
        const previousGrid = this.grid.map(row => [...row]);

        if (direction === 'left' || direction === 'right') {
          for (let r = 0; r < this.size; r++) {
            let row = [...this.grid[r]];
            if (direction === 'right') row.reverse();

            const { newRow, points } = this.slide(row);
            if (direction === 'right') newRow.reverse();

            if (row.toString() !== (direction === 'right' ? [...newRow].reverse() : newRow).toString()) {
              moved = true;
            }
            this.grid[r] = newRow;
            gainedPoints += points;
          }
        } else if (direction === 'up' || direction === 'down') {
          for (let c = 0; c < this.size; c++) {
            let col = [this.grid[0][c], this.grid[1][c], this.grid[2][c], this.grid[3][c]];
            if (direction === 'down') col.reverse();

            const { newRow, points } = this.slide(col);
            if (direction === 'down') newRow.reverse();

            for (let r = 0; r < this.size; r++) {
              if (this.grid[r][c] !== newRow[r]) moved = true;
              this.grid[r][c] = newRow[r];
            }
            gainedPoints += points;
          }
        }

        if (moved) {
          // Push previous state for undo capability
          this.history.push({
            grid: previousGrid,
            score: this.score,
            won: this.won,
            keepPlaying: this.keepPlaying
          });
          this.updateUndoButton();

          if (gainedPoints > 0) {
            this.addScore(gainedPoints);
          }

          const spawnedTile = this.addRandomTile();
          this.render(spawnedTile);

          // Check Win (2048 reached)
          if (!this.won && !this.keepPlaying && this.hasTile(2048)) {
            this.won = true;
            this.winModal.classList.remove('hidden');
          }

          // Check Loss
          if (this.checkGameOver()) {
            this.gameOver = true;
            this.gameOverModal.classList.remove('hidden');
          }

          return true;
        }

        return false;
      }

      hasTile(val) {
        return this.grid.some(row => row.includes(val));
      }

      checkGameOver() {
        // Check for empty cells
        for (let r = 0; r < this.size; r++) {
          for (let c = 0; c < this.size; c++) {
            if (this.grid[r][c] === 0) return false;
            // Check adjacent horizontal
            if (c < this.size - 1 && this.grid[r][c] === this.grid[r][c + 1]) return false;
            // Check adjacent vertical
            if (r < this.size - 1 && this.grid[r][c] === this.grid[r + 1][c]) return false;
          }
        }
        return true;
      }

      addScore(amount) {
        this.score += amount;
        this.scoreEl.textContent = this.score;

        if (this.score > this.bestScore) {
          this.bestScore = this.score;
          this.bestScoreEl.textContent = this.bestScore;
          localStorage.setItem('2048_best_score', this.bestScore.toString());
        }

        // Float-up score notification
        const floatEl = document.createElement('div');
        floatEl.className = 'score-addition';
        floatEl.textContent = `+${amount}`;
        this.scoreFloatContainer.appendChild(floatEl);
        setTimeout(() => floatEl.remove(), 600);
      }

      render(newSpawned = null) {
        this.tileLayer.innerHTML = '';
        this.scoreEl.textContent = this.score;

        // Calculate positions dynamically based on CSS grid dimensions
        const boardEl = document.getElementById('board-container');
        const boardRect = boardEl.getBoundingClientRect();
        const gap = 12;
        const cellSize = (boardRect.width - (gap * 5)) / 4;

        for (let r = 0; r < this.size; r++) {
          for (let c = 0; c < this.size; c++) {
            const val = this.grid[r][c];
            if (val !== 0) {
              const tile = document.createElement('div');
              const tileClass = val <= 2048 ? `tile-${val}` : 'tile-super';
              tile.className = `tile ${tileClass}`;
              tile.textContent = val;

              // Absolute pixel positioning to avoid layout shift
              const leftPos = gap + c * (cellSize + gap);
              const topPos = gap + r * (cellSize + gap);
              tile.style.width = `${cellSize}px`;
              tile.style.height = `${cellSize}px`;
              tile.style.transform = `translate(${leftPos}px, ${topPos}px)`;

              // Mark newly spawned tile for animation
              if (newSpawned && newSpawned.r === r && newSpawned.c === c) {
                tile.classList.add('tile-new');
              }

              this.tileLayer.appendChild(tile);
            }
          }
        }
      }

      initEventListeners() {
        // Keyboard controls
        window.addEventListener('keydown', (e) => {
          if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space'].includes(e.code)) {
            e.preventDefault();
          }

          switch (e.code) {
            case 'ArrowLeft':
            case 'KeyA':
              this.move('left');
              break;
            case 'ArrowRight':
            case 'KeyD':
              this.move('right');
              break;
            case 'ArrowUp':
            case 'KeyW':
              this.move('up');
              break;
            case 'ArrowDown':
            case 'KeyS':
              this.move('down');
              break;
          }
        });

        // Touch swipe handling
        let touchStartX = 0;
        let touchStartY = 0;
        const touchThreshold = 35;

        const board = document.getElementById('board-container');
        board.addEventListener('touchstart', (e) => {
          if (e.touches.length === 1) {
            touchStartX = e.touches[0].clientX;
            touchStartY = e.touches[0].clientY;
          }
        }, { passive: true });

        board.addEventListener('touchend', (e) => {
          if (e.changedTouches.length === 1) {
            const deltaX = e.changedTouches[0].clientX - touchStartX;
            const deltaY = e.changedTouches[0].clientY - touchStartY;
            const absX = Math.abs(deltaX);
            const absY = Math.abs(deltaY);

            if (Math.max(absX, absY) > touchThreshold) {
              if (absX > absY) {
                // Horizontal swipe
                if (deltaX > 0) this.move('right');
                else this.move('left');
              } else {
                // Vertical swipe
                if (deltaY > 0) this.move('down');
                else this.move('up');
              }
            }
          }
        }, { passive: true });

        // Buttons
        document.getElementById('restart-btn').addEventListener('click', () => this.restart());
        document.getElementById('try-again-btn').addEventListener('click', () => this.restart());
        document.getElementById('win-new-game-btn').addEventListener('click', () => this.restart());
        this.undoBtn.addEventListener('click', () => this.undo());

        document.getElementById('keep-going-btn').addEventListener('click', () => {
          this.keepPlaying = true;
          this.winModal.classList.add('hidden');
        });

        // Re-render positions on window resize
        window.addEventListener('resize', () => {
          this.render();
        });
      }
    }

    // Start game when page is loaded
    window.addEventListener('load', () => {
